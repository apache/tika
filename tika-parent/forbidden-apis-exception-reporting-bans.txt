# TIKA-4848: the ExceptionReporting policy reaches a parser only through the ParseContext, so
# the pre-4.1 overloads that take no context silently report FULL. They stay for compatibility;
# these bans keep in-tree code off them. Default-on for main sources in every module
# (see the forbiddenapis-check execution in tika-parent); the deprecated overloads' own
# bodies delegate to the context overloads, so tika-core passes too.

org.apache.tika.utils.ExceptionUtils#getStackTrace(java.lang.Throwable) @ use format(Throwable, ParseContext) so the exception-reporting policy applies
org.apache.tika.utils.ExceptionUtils#getFilteredStackTrace(java.lang.Throwable) @ use format(Throwable, ParseContext) so the exception-reporting policy applies
org.apache.tika.extractor.EmbeddedDocumentUtil#recordException(java.lang.Throwable,org.apache.tika.metadata.Metadata) @ use the ParseContext overload so the exception-reporting policy applies
org.apache.tika.extractor.EmbeddedDocumentUtil#recordEmbeddedStreamException(java.lang.Throwable,org.apache.tika.metadata.Metadata) @ use the ParseContext overload so the exception-reporting policy applies
org.apache.tika.utils.ParserUtils#recordParserFailure(org.apache.tika.parser.Parser,java.lang.Throwable,org.apache.tika.metadata.Metadata) @ use the ParseContext overload so the exception-reporting policy applies
